-- ========================================================
-- BUAT DAN GUNAKAN DATABASE
-- ========================================================
DROP DATABASE IF EXISTS db_penjualan;
CREATE DATABASE db_penjualan;
USE db_penjualan;

-- ========================================================
-- 1. PEMBUATAN TABEL MASTER
-- ========================================================

CREATE TABLE supplier (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nama VARCHAR(100),
    telp VARCHAR(12),
    alamat TEXT
);

CREATE TABLE pelanggan (
    id VARCHAR(20) PRIMARY KEY,
    nama VARCHAR(100),
    jenis_kelamin ENUM('L','P'),
    telp VARCHAR(12),
    alamat TEXT
);

CREATE TABLE user (
    id_user TINYINT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(30),
    password VARCHAR(35),
    nama VARCHAR(50),
    alamat VARCHAR(150),
    hp VARCHAR(20),
    level TINYINT(1)
);

CREATE TABLE barang (
    id INT PRIMARY KEY AUTO_INCREMENT,
    kode_barang VARCHAR(10),
    nama_barang VARCHAR(100),
    harga INT,
    stok INT,
    supplier_id INT,
    FOREIGN KEY (supplier_id) REFERENCES supplier(id)
);

-- ========================================================
-- 2. PEMBUATAN TABEL TRANSAKSI (MASTER-DETAIL)
-- ========================================================

CREATE TABLE transaksi (
    id INT PRIMARY KEY AUTO_INCREMENT,
    waktu_transaksi DATE,
    keterangan TEXT,
    total INT,
    pelanggan_id VARCHAR(20),
    FOREIGN KEY (pelanggan_id) REFERENCES pelanggan(id)
);

CREATE TABLE transaksi_detail (
    transaksi_id INT,
    barang_id INT,
    harga INT,
    qty INT,
    FOREIGN KEY (transaksi_id) REFERENCES transaksi(id),
    FOREIGN KEY (barang_id) REFERENCES barang(id)
);

CREATE TABLE pembayaran (
    id INT PRIMARY KEY AUTO_INCREMENT,
    waktu_bayar DATETIME,
    total INT,
    metode ENUM('TUNAI', 'TRANSFER', 'EDC'),
    transaksi_id INT,
    FOREIGN KEY (transaksi_id) REFERENCES transaksi(id)
);
